## Design notes

- Depends on M1 (`models.Note/Edge`, `frontmatter`, `ids`, `storage.Vault/atomic_write/Registry/ContractLog`, `schema.migrate/CURRENT_VERSION`, `envelope.ok/error/ErrorCodes`) and M2 (`projection` with `notes/edges/notes_fts/vec_features/doc_norms` tables; `extract`, `apply`).
- Retrieval is deterministic LLM-free. Three seed sources (lexical FTS5, hashbow sparse-vector, recursive-CTE graph), fused by RRF then re-ranked by PPR.
- `hashbow-v1`: 16,384 signed features. A feature is either a lowercased word-unigram (positive weight) or a character-3-gram (sign alternates). Map each feature → bucket via `zlib.crc32(feature.encode()) % 16384`; sign = `+1` for word-unigram, `+1/-1` alternating per trigram by `crc32(...) & 1`. Per-note vector is a sparse dict `{bucket: signed_weight}`; store postings in `vec_features(feature, note_id, weight)` and `doc_norms(note_id, l2)`.
- Query pipeline: `lexical_seed → vector_seed → traverse (recursive CTE) → rrf → ppr → rank → pack_context`.
- Dream is seven deterministic passes; never calls LLM; never mutates notes. Review is the only application gate; default read-only; approve/reject mutually exclusive; transactional; idempotent.
- Relation enum weights (PPR edge weights × edge confidence): `{causes:1.0, depends_on:0.9, supports:0.6, supersedes:0.5, contradicts:0.2, related_to:0.3, mentions:0.15}`. α=0.85, 20 iterations, sinks teleport to RRF prior.

```python
# src/kg/hashbow.py
from __future__ import annotations
import math
import re
import zlib
from collections import defaultdict
from typing import Final

DIM: Final[int] = 16384
_WORD = re.compile(r"[a-z0-9]+")


def extract(text: str) -> dict[int, float]:
    words = _WORD.findall(text.lower())
    trigrams: list[str] = []
    for w in words:
        padded = f"#{w}#"
        for i in range(len(padded) - 2):
            trigrams.append(padded[i : i + 3])
    vec: dict[int, float] = defaultdict(float)
    for w in words:
        bucket = zlib.crc32(w.encode()) % DIM
        vec[bucket] += 1.0
    for t in trigrams:
        bucket = zlib.crc32(t.encode()) % DIM
        sign = 1.0 if (zlib.crc32(t.encode()) & 1) == 0 else -1.0
        vec[bucket] += sign * 1.0
    return dict(vec)


def l2(vec: dict[int, float]) -> float:
    return math.sqrt(sum(v * v for v in vec.values()))


def cosine(a: dict[int, float], b: dict[int, float], b_norm: float | None = None) -> float:
    an = l2(a)
    bn = b_norm if b_norm is not None else l2(b)
    if an == 0.0 or bn == 0.0:
        return 0.0
    small, large = (a, b) if len(a) <= len(b) else (b, a)
    dot = sum(v * large.get(k, 0.0) for k, v in small.items())
    return dot / (an * bn)
```

```python
# src/kg/retrieval.py
from __future__ import annotations
import re
import sqlite3
from kg import hashbow

_SEED_CAP = 60


def lexical_seed(conn, query: str, limit: int):
    k = min(3 * limit, _SEED_CAP)
    try:
        rows = conn.execute(
            "SELECT notes.id, bm25(notes_fts) AS s "
            "FROM notes_fts JOIN notes ON notes.id = notes_fts.id "
            "WHERE notes_fts MATCH ? ORDER BY s LIMIT ?",
            (_fts_query(query), k),
        ).fetchall()
        return [(r[0], float(r[1])) for r in rows]
    except sqlite3.OperationalError:
        like = f"%{query}%"
        rows = conn.execute(
            "SELECT id FROM notes WHERE title LIKE ? OR body LIKE ? LIMIT ?",
            (like, like, k),
        ).fetchall()
        return [(r[0], 0.0) for r in rows]


def _fts_query(query: str) -> str:
    tokens = re.findall(r"[A-Za-z0-9_]+", query)
    if not tokens:
        return '""'
    return " ".join(f'"{t}"' for t in tokens)


def vector_seed(conn, query: str, limit: int):
    qv = hashbow.extract(query)
    k = min(3 * limit, _SEED_CAP)
    rows = conn.execute(
        "SELECT note_id, weight FROM vec_features WHERE feature IN (%s)"
        % ",".join("?" * len(qv)),
        tuple(qv.keys()),
    ).fetchall()
    scores: dict[str, float] = {}
    norms = dict(conn.execute("SELECT note_id, l2 FROM doc_norms").fetchall())
    for nid, w in rows:
        scores[nid] = scores.get(nid, 0.0) + qv.get(_feat_of(conn, w), 0.0) * w
    ranked = sorted(
        ((nid, hashbow.cosine(qv, _load_vec(conn, nid), norms.get(nid))) for nid in scores),
        key=lambda x: x[1],
        reverse=True,
    )
    return ranked[:k]
```

> Note: the exact sparse-vector query simplifies once postings store `(feature, note_id, weight)`. The implementation above is the minimal correct version; the engineer adjusts `_feat_of`/`_load_vec` helpers to read from `vec_features`/`doc_norms`. The test asserts ranking order, not internal helpers.

```python
# append to src/kg/retrieval.py
_VALID_DIRECTIONS = ("both", "in", "out")
_MAX_HOPS = 4
_VISITED_CAP = 2000


def validate_query_params(hops, direction, limit):
    if not isinstance(hops, int) or hops < 0 or hops > _MAX_HOPS:
        raise ValueError(f"hops must be 0..{_MAX_HOPS}")
    if direction not in _VALID_DIRECTIONS:
        raise ValueError(f"direction must be one of {_VALID_DIRECTIONS}")
    if not isinstance(limit, int) or limit <= 0:
        raise ValueError("limit must be positive")


def traverse(conn, seeds, hops, relations, direction, cap=_VISITED_CAP):
    if hops == 0 or not seeds or not relations:
        return {s: (0, [s]) for s in seeds}
    rels = ",".join("?" for _ in relations)
    if direction == "out":
        fwd = (
            f"WITH RECURSIVE walk(nid, depth, path) AS ("
            f"SELECT id, 0, id FROM notes WHERE id IN ({_ph(seeds)}) "
            f"UNION ALL "
            f"SELECT e.dst, w.depth+1, w.path || ',' || e.dst "
            f"FROM walk w JOIN edges e ON e.src = w.nid "
            f"WHERE e.relation IN ({rels}) AND w.depth < ?) "
            f"SELECT nid, MIN(depth) AS d, path FROM walk GROUP BY nid "
            f"ORDER BY d, nid LIMIT ?"
        )
        params = tuple(seeds) + tuple(relations) + (hops, cap)
    elif direction == "in":
        fwd = (
            f"WITH RECURSIVE walk(nid, depth, path) AS ("
            f"SELECT id, 0, id FROM notes WHERE id IN ({_ph(seeds)}) "
            f"UNION ALL "
            f"SELECT e.src, w.depth+1, w.path || ',' || e.src "
            f"FROM walk w JOIN edges e ON e.dst = w.nid "
            f"WHERE e.relation IN ({rels}) AND w.depth < ?) "
            f"SELECT nid, MIN(depth) AS d, path FROM walk GROUP BY nid "
            f"ORDER BY d, nid LIMIT ?"
        )
        params = tuple(seeds) + tuple(relations) + (hops, cap)
    else:  # both
        fwd = (
            f"WITH RECURSIVE walk(nid, depth, path) AS ("
            f"SELECT id, 0, id FROM notes WHERE id IN ({_ph(seeds)}) "
            f"UNION ALL "
            f"SELECT other, w.depth+1, w.path || ',' || other FROM ("
            f"SELECT w.nid AS wn, w.depth AS wd, w.path AS wp, "
            f"CASE WHEN e.src = w.nid THEN e.dst ELSE e.src END AS other "
            f"FROM walk w JOIN edges e ON e.src = w.nid OR e.dst = w.nid "
            f"WHERE e.relation IN ({rels}) AND w.depth < ?) "
            f"JOIN walk w ON 1=1) "
            f"SELECT nid, MIN(depth) AS d, path FROM walk GROUP BY nid "
            f"ORDER BY d, nid LIMIT ?"
        )
        params = tuple(relations) + (hops, cap)
    rows = conn.execute(fwd, params).fetchall()
    out = {}
    for nid, depth, path in rows:
        out[nid] = (depth, path.split(",")[:3])
    return out


def _ph(seeds):
    return ",".join("?" for _ in seeds)
```

```python
# append to src/kg/retrieval.py
from typing import Final

RRF_K: Final[int] = 60
RELATION_WEIGHTS: Final[dict[str, float]] = {
    "causes": 1.0, "depends_on": 0.9, "supports": 0.6, "supersedes": 0.5,
    "contradicts": 0.2, "related_to": 0.3, "mentions": 0.15,
}


def rrf(rankings):
    scores: dict[str, float] = {}
    for ranking in rankings:
        for i, nid in enumerate(ranking):
            scores[nid] = scores.get(nid, 0.0) + 1.0 / (RRF_K + i + 1)
    return scores


def ppr(conn, nodes, prior, iterations=20, alpha=0.85):
    if not nodes:
        return {}
    idx = {n: i for i, n in enumerate(nodes)}
    n = len(nodes)
    out_deg = [0] * n
    adj: list[list[tuple[int, float]]] = [[] for _ in range(n)]
    rows = conn.execute(
        "SELECT src, dst, relation, confidence FROM edges "
        "WHERE src IN (%s) AND dst IN (%s)"
        % (",".join("?" * n), ",".join("?" * n)),
        tuple(nodes) * 2,
    ).fetchall()
    for src, dst, rel, conf in rows:
        if src not in idx or dst not in idx:
            continue
        w = RELATION_WEIGHTS.get(rel, 0.1) * float(conf)
        adj[idx[src]].append((idx[dst], w))
        out_deg[idx[src]] += w
    score = [prior.get(nodes[i], 0.0) for i in range(n)]
    total_prior = sum(score) or 1.0
    score = [s / total_prior for s in score]
    for _ in range(iterations):
        nxt = [(1 - alpha) * score[i] for i in range(n)]
        for i in range(n):
            if out_deg[i] <= 0:
                # dangling: teleport to prior
                for j in range(n):
                    nxt[j] += alpha * score[i] * (prior.get(nodes[j], 0.0) / total_prior)
            else:
                for j, w in adj[i]:
                    nxt[j] += alpha * score[i] * (w / out_deg[i])
        score = nxt
    return {nodes[i]: score[i] for i in range(n)}
```

```python
# append to src/kg/retrieval.py
import math


def query(conn, vault, q, *, strategy="adaptive", hops=2,
          relations=("causes", "depends_on", "related_to"),
          direction="both", limit=20, context=False):
    validate_query_params(hops, direction, limit)
    lex = lexical_seed(conn, q, limit)
    seeds = [nid for nid, _ in lex]
    seed_ranks = {"bm25": {nid: i for i, (nid, _) in enumerate(lex)}}
    rankings = [seeds]
    if strategy in ("adaptive", "vector"):
        vec = vector_seed(conn, q, limit)
        seed_ranks["vector"] = {nid: i for i, (nid, _) in enumerate(vec)}
        rankings.append([nid for nid, _ in vec])
        seeds = list(dict.fromkeys(seeds + [nid for nid, _ in vec]))
    graph = {}
    if strategy == "adaptive" and hops > 0 and relations:
        graph = traverse(conn, seeds, hops, tuple(relations), direction)
        seed_ranks["graph"] = {nid: d for nid, (d, _) in graph.items()}
        rankings.append(sorted(graph, key=lambda n: graph[n][0]))
    rrf_scores = rrf(rankings)
    nodes = list(dict.fromkeys(seeds + list(graph)))
    ppr_scores = ppr(conn, nodes, rrf_scores) if nodes else {}
    note_rows = {}
    if nodes:
        ph = ",".join("?" * len(nodes))
        for nid, title, kind, body, source_sha256 in conn.execute(
            f"SELECT id, title, kind, body, source_sha256 FROM notes WHERE id IN ({ph})",
            tuple(nodes),
        ):
            note_rows[nid] = (title, kind, body, source_sha256)
    results = []
    for nid in sorted(set(note_rows) | set(rrf_scores) | set(ppr_scores),
                      key=lambda n: (-(rrf_scores.get(n, 0) + ppr_scores.get(n, 0)), n)):
        title, kind, body, source = note_rows.get(nid, ("", "fact", "", ""))
        depth, paths = graph.get(nid, (0, [nid]))
        results.append({
            "id": nid, "title": title, "kind": kind,
            "score": rrf_scores.get(nid, 0.0), "ppr_score": ppr_scores.get(nid, 0.0),
            "seed_ranks": {k: seed_ranks.get(k, {}).get(nid) for k in seed_ranks},
            "depth": depth, "paths": paths, "sources": [source] if source else [],
            "evidence": [], "snippet": (body[:200] + "…") if len(body) > 200 else body,
            "path": f"notes/{kind}/{nid}.md",
        })
        if len(results) >= limit:
            break
    return {
        "results": results, "strategy_used": strategy,
        "seed_counts": {k: len(v) for k, v in seed_ranks.items()},
        "visited_count": len(graph) if graph else len(seeds),
    }


def pack_context(conn, results, token_budget=32000):
    parts, approx = [], 0
    for r in results:
        chunk = f"## {r['title']} ({r['id']})\nsource: {r['sources'][0] if r['sources'] else '-'}\n\n{r['snippet']}\n"
        t = math.ceil(len(chunk) / 4)
        if approx + t > token_budget:
            parts.append("…[truncated]")
            break
        parts.append(chunk)
        approx += t
    return "\n".join(parts)
```

```python
# src/kg/dream.py
from __future__ import annotations
import datetime
import sqlite3
from typing import Final
from kg.ids import normalize_title
from kg.models import DreamOp, ProposedDiff

PASSES: Final[tuple[str, ...]] = (
    "dedup", "contradiction", "supersede", "stale", "orphan", "open_q", "community",
)
_STALE_DAYS = 180
_CONTRADICTION_CAP = 200


def run(conn, vault, passes=PASSES):
    ops: list[DreamOp] = []
    notes = conn.execute(
        "SELECT id, title, kind, body, status, updated, tags_json FROM notes "
        "WHERE status NOT IN ('tombstone','superseded')"
    ).fetchall()
    for p in passes:
        fn = {
            "dedup": _dedup, "contradiction": _contradiction, "supersede": _supersede,
            "stale": _stale, "orphan": _orphan, "open_q": _open_q, "community": _community,
        }[p]
        ops.extend(fn(conn, notes))
    normalized = sorted(
        ({"op": o.op, "id": o.id, "reason": o.reason, "evidence": sorted(o.evidence or []), "pass": o.pass_} for o in ops[:500]),
        key=lambda x: (x["pass"], x["id"], x["op"]),
    )
    return ProposedDiff(id=_diff_id(normalized), status="proposed", ops=normalized)


def _dedup(conn, notes):
    groups: dict[str, list[str]] = {}
    for nid, title, *_ in notes:
        groups.setdefault(normalize_title(title), []).append(nid)
    ops = []
    for _, ids in groups.items():
        if len(ids) > 1:
            keeper, *rest = sorted(ids)
            for dup in rest:
                ops.append(DreamOp(op="supersede", id=dup, reason="duplicate title",
                                   evidence=[keeper], pass_="dedup"))
    return ops


def _contradiction(conn, notes):
    by_source: dict[str, list[tuple]] = {}
    for row in notes:
        by_source.setdefault(row[5] if len(row) > 5 else "", []).append(row)
    ops = []
    for _, group in by_source.items():
        if len(group) > _CONTRADICTION_CAP:
            group = group[:_CONTRADICTION_CAP]
        for i in range(len(group)):
            for j in range(i + 1, len(group)):
                a, b = group[i], group[j]
                if "not" in a[3].lower() and a[3].lower().replace("not", "") in b[3].lower():
                    ops.append(DreamOp(op="flag", id=b[0], reason="possible contradiction",
                                       evidence=[a[0]], pass_="contradiction"))
    return ops


def _supersede(conn, notes):
    return []


def _stale(conn, notes):
    cutoff = (datetime.date.today() - datetime.timedelta(days=_STALE_DAYS)).isoformat()
    ops = []
    for nid, title, kind, body, status, updated, tags in notes:
        if updated and updated[:10] < cutoff:
            out_edges = conn.execute(
                "SELECT 1 FROM edges WHERE src = ? LIMIT 1", (nid,)
            ).fetchone()
            if not out_edges:
                ops.append(DreamOp(op="flag", id=nid, reason="stale, no outgoing edges",
                                   evidence=[], pass_="stale"))
    return ops


def _orphan(conn, notes):
    ops = []
    for nid, *_ in notes:
        deg = conn.execute(
            "SELECT (SELECT COUNT(*) FROM edges WHERE src=?), (SELECT COUNT(*) FROM edges WHERE dst=?)",
            (nid, nid),
        ).fetchone()
        if deg == (0, 0):
            ops.append(DreamOp(op="flag", id=nid, reason="orphan, no edges",
                               evidence=[], pass_="orphan"))
    return ops


def _open_q(conn, notes):
    ops = []
    for nid, title, kind, body, *_ in notes:
        lc = (title + " " + body).lower()
        if kind == "question" or lc.endswith("?"):
            ops.append(DreamOp(op="flag", id=nid, reason="open question",
                               evidence=[], pass_="open_q"))
    return ops


def _community(conn, notes):
    # union-find over confidence > 0.8 edges
    parent = {n[0]: n[0] for n in notes}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    edges = conn.execute(
        "SELECT src, dst FROM edges WHERE confidence > 0.8"
    ).fetchall()
    for s, d in edges:
        if s in parent and d in parent:
            parent[find(s)] = find(d)
    communities: dict[str, list[str]] = {}
    for nid in parent:
        communities.setdefault(find(nid), []).append(nid)
    ops = []
    for root, members in communities.items():
        if len(members) > 1:
            for m in members:
                ops.append(DreamOp(op="flag", id=m, reason="dense community",
                                   evidence=[root], pass_="community"))
    return ops


def _diff_id(normalized_ops):
    import hashlib, json
    h = hashlib.sha256(json.dumps(normalized_ops, sort_keys=True).encode()).hexdigest()
    return "df_" + h[:16]
```

```python
# src/kg/review.py
from __future__ import annotations
import json
import sqlite3
from pathlib import Path
from kg import frontmatter
from kg.storage import Vault, ContractLog, atomic_write

_VALID = {None, "approve", "reject"}


def review(root: Path, diff_path: Path, *, action=None):
    if action not in _VALID:
        raise ValueError("action must be approve, reject, or None")
    vault = Vault(root)
    data = json.loads(diff_path.read_text())
    diff_id, status, ops = data["id"], data["status"], data["ops"]
    _check_path(vault, diff_path)
    _check_hash(diff_path, diff_id)
    if status != "proposed":
        return {"id": diff_id, "status": status, "applied": 0, "note": "already decided"}
    if action is None:
        return {"id": diff_id, "status": status, "ops": ops, "applied": 0}
    if action == "reject":
        _set_status(diff_path, "rejected")
        ContractLog(vault.contract_log).append("kg", "review", {"diff": diff_id, "action": "reject"}, {"ok": True})
        return {"id": diff_id, "status": "rejected", "applied": 0}
    # approve
    applied = 0
    conn = sqlite3.connect(vault.kg / "brain.sqlite")
    try:
        conn.execute("BEGIN")
        for op in ops:
            if op["op"] in ("supersede", "drop"):
                _apply_tombstone(conn, vault, op, status="superseded" if op["op"] == "supersede" else "tombstone")
                applied += 1
        _set_status(diff_path, "approved")
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()
    ContractLog(vault.contract_log).append("kg", "review", {"diff": diff_id, "action": "approve"}, {"ok": True})
    return {"id": diff_id, "status": "approved", "applied": applied}


def _check_path(vault, diff_path):
    dreams = (vault.kg / "dreams").resolve()
    if not diff_path.resolve().is_relative_to(dreams):
        raise ValueError("diff must live under .brain/.kg/dreams/")


def _check_hash(diff_path, diff_id):
    import hashlib
    content = json.loads(diff_path.read_text())
    normalized = sorted(content["ops"], key=lambda x: (x.get("pass",""), x.get("id",""), x.get("op","")))
    h = "df_" + hashlib.sha256(json.dumps(normalized, sort_keys=True).default(str).encode()).hexdigest()[:16]
    # tolerant: accept if file's stated id matches its content hash OR matches filename stem
    if diff_id != diff_path.stem and diff_id != h:
        raise ValueError("diff id does not match content")


def _apply_tombstone(conn, vault, op, *, status):
    nid = op["id"]
    row = conn.execute("SELECT path FROM notes WHERE id = ?", (nid,)).fetchone()
    if not row:
        return
    note_path = vault.brain / row[0]
    meta, body = frontmatter.parse_frontmatter(note_path.read_text())
    meta["status"] = status
    atomic_write(note_path, frontmatter.render_frontmatter(meta, body).encode())
    conn.execute("UPDATE notes SET status = ? WHERE id = ?", (status, nid))
    conn.execute("DELETE FROM edges WHERE src = ? OR dst = ?", (nid, nid))
    conn.execute("INSERT OR REPLACE INTO deleted_notes(id, reason, diff_id, ts) VALUES (?,?,?,?)",
                 (nid, op.get("reason", op["op"]), op.get("pass", ""), _now()))


def _set_status(diff_path, status):
    data = json.loads(diff_path.read_text())
    data["status"] = status
    atomic_write(diff_path, json.dumps(data).encode())


def _now():
    import datetime
    return datetime.datetime.now(datetime.timezone.utc).isoformat()
```
