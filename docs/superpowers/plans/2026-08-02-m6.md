# Milestone M6 Implementation Plan

> Spec: `docs/superpowers/specs/2026-08-02-unified-brain-kg-design.md`. Existing scaffold is rejected; rewrite. TDD bite-sized steps. Each task: Files → Interfaces → steps (failing test, verify-fail run, impl, verify-pass run, commit).

## Design notes

- Rewrite rejected scaffold; retain only interfaces delivered by M1–M5.
- Distribution remains Python-only; no Node runtime, MCP, CDN, daemon, harness mutation, extra graph formats, or unpinned UI.
- Release version is 1.0.0; package metadata, kg --version, wheel metadata, installer expectations, and CI checks must agree.
- Installer verifies pinned source and wheel hashes before uv tool install; benchmark release gate rejects absent or NOT_MEASURED required cells.
- Every task follows test-first development, independent verification, and a commit.

```python
from __future__ import annotations

import json
import os
import subprocess
import time
from pathlib import Path
from typing import Mapping

ReportCell = dict[str, object]


def run_manifest(manifest_path: Path, report_path: Path, harness_commands: dict[str, list[str]], env: Mapping[str, str] | None = None) -> list[ReportCell]:
    manifest = json.loads(manifest_path.read_text())
    current_env = dict(os.environ if env is None else env)
    cells: list[ReportCell] = []
    for harness, command in harness_commands.items():
        credential = {'claude-code': 'CLAUDE_API_KEY', 'cursor': 'CURSOR_API_KEY', 'pi': 'PI_API_KEY'}.get(harness)
        for task in manifest['tasks']:
            if credential and not current_env.get(credential):
                cells.append({'harness': harness, 'task': task['id'], 'pass': None, 'time_s': 0.0, 'tokens': None, 'cost': None, 'status': 'NOT_MEASURED', 'reason': f'missing credential: {credential}'})
                continue
            started = time.monotonic()
            result = subprocess.run(command + [task['prompt']], text=True, capture_output=True, check=False, timeout=int(task['timeout_s']), shell=False, env=current_env)
            elapsed = round(time.monotonic() - started, 3)
            try:
                payload = json.loads(result.stdout)
            except json.JSONDecodeError:
                payload = {}
            cells.append({'harness': harness, 'task': task['id'], 'pass': bool(payload.get('pass')) if result.returncode == 0 else False, 'time_s': elapsed, 'tokens': payload.get('tokens'), 'cost': payload.get('cost'), 'status': 'MEASURED' if result.returncode == 0 else 'FAILED', 'reason': '' if result.returncode == 0 else result.stderr.strip() or f'exit code {result.returncode}'})
    report_path.write_text(json.dumps(cells, sort_keys=True, indent=2) + '\n')
    return cells
PY
```

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Iterable, Mapping, Sequence


def validate_release_cells(cells: Iterable[Mapping[str, object]], required_harnesses: Sequence[str], required_tasks: Sequence[str]) -> None:
    indexed = {(str(cell.get('harness')), str(cell.get('task'))): cell for cell in cells}
    errors: list[str] = []
    for harness in required_harnesses:
        for task in required_tasks:
            key = (harness, task)
            cell = indexed.get(key)
            if cell is None:
                errors.append(f'absent: {harness}/{task}')
                continue
            if cell.get('status') == 'NOT_MEASURED':
                errors.append(f'NOT_MEASURED: {harness}/{task}: {cell.get("reason", "")}'.rstrip())
            elif cell.get('status') != 'MEASURED' or cell.get('pass') is not True:
                errors.append(f'failed: {harness}/{task}')
    if errors:
        raise ValueError('; '.join(errors))


def gate_report(path: Path, required_harnesses: Sequence[str], required_tasks: Sequence[str]) -> None:
    validate_release_cells(json.loads(path.read_text()), required_harnesses, required_tasks)
PY
```
